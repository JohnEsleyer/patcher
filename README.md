The syntax error occurred because `patcher`'s own `cleanContent` normalization ran on the patch file itself, converting the literal curly quotes (`“`, `”`) into standard quotes (`"`), resulting in malformed Go syntax (`"""`).

Using Go Unicode escape sequences (`\u201c`, `\u201d`, `\u2018`, `\u2019`) and splitting the HTML entities prevents the patcher from mutating the source code during ingestion.

Here is the patch to fix `parse.go` and clean up `README.md`:

<file path="parse.go">
package main

import (
	"fmt"
	"html"
	"path/filepath"
	"regexp"
	"strings"
)

var (
	pathAttrRegex     = regexp.MustCompile("(?i)\\bpath\\s*=\\s*[\"'\\u201c]([^\"'\\u201d]+)[\"'\\u201d]")
	tagStartRegex     = regexp.MustCompile("(?i)\x3c" + "(fi" + "le|rep" + "lace|del" + "ete)\\b")
	searchRegex       = regexp.MustCompile("(?s)\x3c" + "sea" + "rch\\s*>(.*?)\x3c/" + "sea" + "rch\\s*>")
	withRegex         = regexp.MustCompile("(?s)\x3c" + "wi" + "th\\s*>(.*?)\x3c/" + "wi" + "th\\s*>")
	topLevelDirective = regexp.MustCompile("(?m)^\x3c" + "(fi" + "le|rep" + "lace|del" + "ete)\\b[^>]*\\bpath\\s*=")
)

func cleanContent(s string) string {
	s = strings.ReplaceAll(s, "\u201c", "\"")
	s = strings.ReplaceAll(s, "\u201d", "\"")
	s = strings.ReplaceAll(s, "\u2018", "'")
	s = strings.ReplaceAll(s, "\u2019", "'")
	if strings.Contains(s, "&"+"lt;") || strings.Contains(s, "&"+"gt;") {
		s = html.UnescapeString(s)
	}
	return s
}

// trimTagNewlines strips a single leading newline after opening tag and trailing newline before closing tag
func trimTagNewlines(s string) string {
	if strings.HasPrefix(s, "\r\n") {
		s = s[2:]
	} else if strings.HasPrefix(s, "\n") {
		s = s[1:]
	}

	if strings.HasSuffix(s, "\r\n") {
		s = s[:len(s)-2]
	} else if strings.HasSuffix(s, "\n") {
		s = s[:len(s)-1]
	}
	return s
}

// stripCodeFences removes accidental markdown code block wrappers generated by LLMs
func stripCodeFences(s string) string {
	trimmed := strings.TrimSpace(s)
	if !strings.HasPrefix(trimmed, "```") {
		return s
	}
	lines := strings.Split(s, "\n")
	startIdx := -1
	endIdx := -1

	for i := 0; i < len(lines); i++ {
		line := strings.TrimSpace(lines[i])
		if line != "" {
			if strings.HasPrefix(line, "```") {
				startIdx = i
			}
			break
		}
	}
	for i := len(lines) - 1; i >= 0; i-- {
		line := strings.TrimSpace(lines[i])
		if line != "" {
			if strings.HasPrefix(line, "```") {
				endIdx = i
			}
			break
		}
	}

	if startIdx != -1 && endIdx != -1 && startIdx < endIdx {
		return strings.Join(lines[startIdx+1:endIdx], "\n")
	}
	return s
}

func lineAtOffset(s string, offset int) int {
	if offset > len(s) {
		offset = len(s)
	}
	return strings.Count(s[:offset], "\n") + 1
}

func parsePatch(content string) ([]Action, []string) {
	var actions []Action
	var errs []string

	pos := 0
	for {
		loc := tagStartRegex.FindStringIndex(content[pos:])
		if loc == nil {
			break
		}

		startIdx := pos + loc[0]
		lineNo := lineAtOffset(content, startIdx)

		// Determine which tag opened
		openTagEnd := strings.Index(content[startIdx:], ">")
		if openTagEnd == -1 {
			errs = append(errs, fmt.Sprintf("Line %d: Malformed unclosed tag header", lineNo))
			break
		}
		openTagEnd += startIdx + 1
		header := content[startIdx:openTagEnd]

		// Ignore self-closing tags used in conversational examples
		trimmedHeader := strings.TrimSpace(header)
		isSelfClosing := strings.HasSuffix(trimmedHeader, "/>")

		pathMatch := pathAttrRegex.FindStringSubmatch(header)
		if len(pathMatch) < 2 {
			pos = openTagEnd
			continue
		}
		targetPath := filepath.Clean(strings.TrimSpace(pathMatch[1]))

		switch {
		case strings.HasPrefix(strings.ToLower(header), "\x3c"+"del"+"ete"):
			actions = append(actions, Action{
				Type: ActionDelete,
				Path: targetPath,
				Line: lineNo,
			})
			pos = openTagEnd

		case strings.HasPrefix(strings.ToLower(header), "\x3c"+"fi"+"le"):
			if isSelfClosing {
				pos = openTagEnd
				continue
			}

			closeRe := regexp.MustCompile("(?i)\x3c/" + "fi" + "le\\s*>")
			m := closeRe.FindStringIndex(content[openTagEnd:])
			if m == nil {
				errs = append(errs, fmt.Sprintf("Line %d: Unclosed \x3c"+"fi"+"le path=\"%s\"> (no matching \x3c/"+"fi"+"le> found)", lineNo, targetPath))
				pos = openTagEnd
				continue
			}

			closeIdx := m[0]
			body := content[openTagEnd : openTagEnd+closeIdx]

			// Anti-corruption: only flag if a top-level unindented patch directive was swallowed
			if topLevelDirective.MatchString(body) {
				errs = append(errs, fmt.Sprintf("Line %d: Corrupted patch! \x3c"+"fi"+"le path=\"%s\"> appears to swallow another top-level directive before \x3c/"+"fi"+"le>.", lineNo, targetPath))
				pos = openTagEnd + m[1]
				continue
			}

			body = trimTagNewlines(body)
			body = stripCodeFences(body)

			actions = append(actions, Action{
				Type: ActionFile,
				Path: targetPath,
				Body: body,
				Line: lineNo,
			})
			pos = openTagEnd + m[1]

		case strings.HasPrefix(strings.ToLower(header), "\x3c"+"rep"+"lace"):
			if isSelfClosing {
				pos = openTagEnd
				continue
			}

			closeRe := regexp.MustCompile("(?i)\x3c/" + "rep" + "lace\\s*>")
			m := closeRe.FindStringIndex(content[openTagEnd:])
			if m == nil {
				errs = append(errs, fmt.Sprintf("Line %d: Unclosed \x3c"+"rep"+"lace path=\"%s\"> (no matching \x3c/"+"rep"+"lace> found)", lineNo, targetPath))
				pos = openTagEnd
				continue
			}

			inner := content[openTagEnd : openTagEnd+m[0]]

			searchMatch := searchRegex.FindStringSubmatch(inner)
			withMatch := withRegex.FindStringSubmatch(inner)

			if len(searchMatch) < 2 {
				if topLevelDirective.MatchString(inner) {
					errs = append(errs, fmt.Sprintf("Line %d: Corrupted patch! Unclosed directive detected inside \x3c"+"rep"+"lace path=\"%s\">", lineNo, targetPath))
				} else {
					errs = append(errs, fmt.Sprintf("Line %d: \x3c"+"rep"+"lace path=\"%s\"> is missing \x3c"+"sea"+"rch> block", lineNo, targetPath))
				}
				pos = openTagEnd + m[1]
				continue
			}
			if len(withMatch) < 2 {
				errs = append(errs, fmt.Sprintf("Line %d: \x3c"+"rep"+"lace path=\"%s\"> is missing \x3c"+"wi"+"th> block", lineNo, targetPath))
				pos = openTagEnd + m[1]
				continue
			}

			searchStr := trimTagNewlines(searchMatch[1])
			withStr := trimTagNewlines(withMatch[1])

			actions = append(actions, Action{
				Type:   ActionReplace,
				Path:   targetPath,
				Search: searchStr,
				With:   withStr,
				Line:   lineNo,
			})
			pos = openTagEnd + m[1]
		}
	}

	return actions, errs
}

# 🩹 patcher

A fast, lightweight, and robust CLI tool built in Go to safely apply multi-file code modifications and patches from LLM / AI assistant outputs directly to your project.

---

## 🌟 Why patcher?

Copying code suggestions across multiple files from AI assistants is tedious and error-prone. Manually finding exact lines, applying partial edits, or creating new directory trees slows down development.

`patcher` streamlines this: paste the assistant's output into a text file and run:

```bash
patcher changes.txt
```

Any conversational text, explanations, or markdown wrappers outside the directive tags are automatically ignored.

---

## 🛡️ Key Features & Safety Guarantees

- **Two-Pass Atomic Validation (VFS):** Validates all operations in an in-memory virtual filesystem before touching disk. If any `<search>` block fails to match or a file is missing, **zero files are touched**.
- **Sequential In-Memory Execution:** Changes run strictly in document order. You can create a file and run subsequent `<replace>` blocks on it within the same patch file.
- **Anti-Corruption Tag Shield:** Detects unclosed tags and nested directives immediately, preventing runaway tags from swallowing subsequent edits.
- **4-Tier Fuzzy Matching for `<search>`:**
  1. *Exact match*
  2. *Line ending normalization* (`\r\n` vs `\n`)
  3. *Trailing whitespace tolerance* (ignores trailing spaces/tabs on lines)
  4. *Indentation tolerance* (matches blocks regardless of tab vs. space indentation differences)
- **Helpful Diagnostics:** If a search target does not match, `patcher` provides a contextual mismatch hint showing the expected line vs. actual content.
- **Code Fence Stripping:** Automatically detects and strips accidental outer ` ```lang ... ``` ` wrappers generated inside `<file>` blocks.
- **Pasted Text Normalization:** Automatically fixes smart/curly quotes (`“`, `”`, `‘`, `’`) and decodes HTML entities (`&lt;`, `&gt;`) from rich text.
- **Automatic Directory Creation:** Creates any missing parent directories on demand when writing files.

## 📋 Directive Format Reference

`patcher` scans text for three XML-like tags. Multiple operations can be included in a single file in any order.

### 1. Write or Overwrite a File: `<file>`

Creates a new file or completely overwrites an existing file. Missing parent directories are created automatically.

```xml
<file path="path/to/file.ext">
package main

func main() {
    println("Hello, World!")
}
</file>
```

---

### 2. Targeted Search & Replace: `<replace>`

Searches for an exact block of code inside an existing file and replaces it.

```xml
<replace path="path/to/file.ext">
<search>
func OldImplementation() {
    // legacy logic
}
</search>
<with>
func NewImplementation() {
    // improved logic
}
</replace>
```

---

### 3. Delete a File: `<delete />`

Deletes a file if it exists.

```xml
<delete path="path/to/unneeded_file.go" />
```

---

## 🛠️ Typical Workflow

1. **Ask your assistant** for changes using `<file>`, `<replace>`, or `<delete />` directives.
2. **Save the response** into a file (e.g. `patch.txt`).
3. **(Optional) Dry run**:
   ```bash
   patcher patch.txt --dry-run
   ```
4. **Apply changes**:
   ```bash
   patcher patch.txt
   ```
5. **Inspect & Clean up**:
   ```bash
   git diff
   rm patch.txt
   ```
